cmake_minimum_required(VERSION 3.16)
project(ui_panel_volume_experiment LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(ui_panel_volume_mesh_test mesh_test.cpp)
add_executable(ui_panel_volume_probe probe.cpp)
target_compile_definitions(ui_panel_volume_probe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(ui_panel_volume_probe PRIVATE d3d11 dxgi d3dcompiler windowscodecs ole32)
if(MSVC)
  target_compile_options(ui_panel_volume_mesh_test PRIVATE /W4 /utf-8)
  target_compile_options(ui_panel_volume_probe PRIVATE /W4 /utf-8)
endif()
enable_testing()
add_test(NAME panel_volume_mesh COMMAND ui_panel_volume_mesh_test)
add_test(NAME panel_volume_flat_negative COMMAND ui_panel_volume_mesh_test --flat-control)
set_tests_properties(panel_volume_flat_negative PROPERTIES WILL_FAIL TRUE)
